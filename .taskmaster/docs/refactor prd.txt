## Advanced Search (with auto-redirect)

`app/search/page.tsx` (UI stub; keeps the redirect behavior):

```tsx
'use client';

import { useEffect, useMemo, useState } from 'react';
import { useRouter, useSearchParams } from 'next/navigation';
import { ListingGrid } from '@/components/listings/ListingGrid';
import SearchBox from '@/components/search/SearchBox';
import FiltersSidebar from '@/components/search/FiltersSidebar';
import type { AppListingCard } from '@/types/appView';

export default function SearchPage() {
  const router = useRouter();
  const params = useSearchParams();

  // derive user selections from QS
  const q = params.get('q') ?? '';
  const cities = params.getAll('city');
  const types = params.getAll('type');
  const amenities = params.getAll('amenity');

  const body = useMemo(() => ({ query: q, cities, types, amenities }), [q, cities, types, amenities]);

  const [listings, setListings] = useState<AppListingCard[]>([]);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    let cancelled = false;
    (async () => {
      setLoading(true);
      try {
        // UI-only: replace with your real fetch later
        const results: AppListingCard[] = [];
        if (!cancelled) setListings(results);

        // auto-redirect if exactly one
        if (!cancelled && results.length === 1 && results[0].slug) {
          router.replace(`/listings/${results[0].slug}`);
        }
      } finally {
        if (!cancelled) setLoading(false);
      }
    })();
    return () => { cancelled = true; };
  }, [body, router]);

  return (
    <div className="container grid grid-cols-1 gap-6 py-8 lg:grid-cols-[280px_1fr]">
      <aside className="rounded-xl border p-4">
        <FiltersSidebar />
      </aside>
      <main className="space-y-4">
        <SearchBox initialValue={q} />
        {loading ? <div>Loading…</div> : <ListingGrid listings={listings} />}
      </main>
    </div>
  );
}
```
